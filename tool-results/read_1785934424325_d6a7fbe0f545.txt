   238→function styleSheet(ws: ExcelJS.Worksheet, locale: Locale) {
   239→  // Header row styling
   240→  const headerRow = ws.getRow(1);
   241→  headerRow.font = { bold: true, size: 11, color: { argb: HEADER_FONT_COLOR } };
   242→  headerRow.fill = { type: "pattern", pattern: "solid", fgColor: { argb: HEADER_FILL } };
   243→  headerRow.alignment = { vertical: "middle", horizontal: "center", wrapText: true };
   244→  headerRow.height = 28;
   245→  headerRow.border = {
   246→    top: { style: "thin", color: { argb: BORDER_COLOR } },
   247→    bottom: { style: "medium", color: { argb: BORDER_COLOR } },
   248→    left: { style: "thin", color: { argb: BORDER_COLOR } },
   249→    right: { style: "thin", color: { argb: BORDER_COLOR } },
   250→  };
   251→
   252→  // Freeze header + enable filter + RTL for Arabic
   253→  const view: Partial<ExcelJS.WorksheetView> = { state: "frozen", ySplit: 1 };
   254→  if (locale === "ar") view.rightToLeft = true;
   255→  ws.views = [view as ExcelJS.WorksheetView];
   256→
   257→  // Auto-filter on the header row
   258→  const colCount = ws.columnCount;
   259→  if (colCount > 0) {
   260→    const lastLetter = ws.getColumn(colCount).letter;
   261→    ws.autoFilter = `A1:${lastLetter}1`;
   262→  }
   263→
   264→  // Page setup — fit-to-width when printing / exporting to PDF
   265→  ws.pageSetup.orientation = "landscape";
   266→  ws.pageSetup.fitToPage = true;
   267→  ws.pageSetup.fitToWidth = 1;
   268→  ws.pageSetup.fitToHeight = 0;
   269→  ws.properties.defaultRowHeight = 18;
   270→}

   272→function autoWidth(ws: ExcelJS.Worksheet, opts: { wrapCols?: Set<number> } = {}) {
   273→  const wrapCols = opts.wrapCols ?? new Set<number>();
   274→  const colCount = ws.columnCount;
   275→  const rowCount = ws.rowCount;
   276→  for (let c = 1; c <= colCount; c++) {
   277→    let maxLen = 0;
   278→    for (let r = 1; r <= rowCount; r++) {
   279→      const cell = ws.getCell(r, c);
   280→      let val = "";
   281→      if (cell.value instanceof Date) {
   282→        val = cell.value.toLocaleDateString();
   283→      } else if (cell.value !== null && cell.value !== undefined) {
   284→        val = String(cell.value);
   285→      }
   286→      if (wrapCols.has(c)) {
   287→        if (val.length > maxLen && val.length <= 50) maxLen = val.length;
   288→      } else {
   289→        const lines = val.split("\n");
   290→        const longest = Math.max(...lines.map((l) => l.length), 0);
   291→        if (longest > maxLen) maxLen = longest;
   292→      }
   293→    }
   294→    ws.getColumn(c).width = Math.min(Math.max(maxLen + 3, 12), 60);
   295→  }
   296→}

   298→function applyDataStyling(ws: ExcelJS.Worksheet, rowCount: number) {
   299→  if (rowCount === 0) return;
   300→  // Apply borders + zebra striping to data rows
   301→  for (let r = 2; r <= rowCount + 1; r++) {
   302→    const row = ws.getRow(r);
   303→    row.border = {
   304→      top: { style: "thin", color: { argb: BORDER_COLOR } },
   305→      bottom: { style: "thin", color: { argb: BORDER_COLOR } },
   306→      left: { style: "thin", color: { argb: BORDER_COLOR } },
   307→      right: { style: "thin", color: { argb: BORDER_COLOR } },
   308→    };
   309→    if (r % 2 === 1) {
   310→      // Light zebra stripe
   311→      row.fill = { type: "pattern", pattern: "solid", fgColor: { argb: "FFF6F8FA" } };
   312→    }
   313→    row.alignment = { vertical: "middle", horizontal: "left", wrapText: false };
   314→  }

   317→function writeIdCell(ws: ExcelJS.Worksheet, row: ExcelJS.Row, colIdx: number, value: string) {
   318→  // National IDs / ref numbers must be text — preserves leading zeros

   324→function writeDateCell(cell: ExcelJS.Cell, value: Date | string | null | undefined, withTime = false) {
   325→  if (!value) {
   326→    cell.value = "";
   327→    return;
   328→  }
   329→  const date = typeof value === "string" ? new Date(value) : value;
   330→  if (Number.isNaN(date.getTime())) {
   331→    cell.value = "";
   332→    return;
   333→  }
   334→  cell.value = date;
   335→  cell.numFmt = withTime ? "yyyy-mm-dd hh:mm" : "yyyy-mm-dd";
   336→}

   340→// ─────────────────────────────────────────────────────────────────────────────
   341→export const GET = async (req: NextRequest) => {
   342→  const user = await requireAuth();
   343→
   344→  // Contractors are scoped to their own company. Coordinators / Super Admin /
   345→  // Company Admin can export ALL companies' data (they don't have companyId).
   346→  const isGlobalRole = ["COORDINATOR", "SUPER_ADMIN", "COMPANY_ADMIN", "TRAINER", "AUDITOR"].includes(user.role);
   347→  if (!user.companyId && !isGlobalRole) {
   348→    return fail("No company linked", 403);
   349→  }
   350→
   351→  const url = new URL(req.url);
   352→  const scope = (url.searchParams.get("scope") || "last") as
   353→    | "last" | "specific_request" | "specific_course" | "date_range" | "all";
   354→  const items = (url.searchParams.get("items") || "").split(",").filter(Boolean);
   355→  const localeParam = url.searchParams.get("locale") === "ar" ? "ar" : "en";
   356→  const locale: Locale = localeParam;
   357→  const specificId = url.searchParams.get("specificId") || undefined;
   358→  const dateFrom = url.searchParams.get("dateFrom") || undefined;
   359→  const dateTo = url.searchParams.get("dateTo") || undefined;
   360→  const format = (url.searchParams.get("format") || "excel") as "excel" | "pdf" | "zip";
   361→
   362→  if (items.length === 0) {
   363→    return fail("No items selected", 422, "VALIDATION_ERROR");
   364→  }
   365→
   366→  const companyId = user.companyId; // undefined for global roles → no company filter
   367→
   368→  // ── Build the TrainingRequest where-clause ──
   369→  // Contractors: scoped to their own company.
   370→  // Coordinators/Admins: see ALL companies (no companyId filter).
   371→  const reqWhere: Record<string, unknown> = { deletedAt: null };
   372→  if (companyId) reqWhere.companyId = companyId;
   373→  let scopeCourseId: string | undefined;
   374→  let scopeCourseTitle = "";
   375→  let scopeRequestRef = "";
   376→
   377→  switch (scope) {
   378→    case "last": {
   379→      const last = await db.trainingRequest.findFirst({
   380→        where: reqWhere,
   381→        orderBy: { createdAt: "desc" },
   382→        select: { id: true, refNumber: true },
   383→      });
   384→      if (last) {
   385→        reqWhere.id = last.id;
   386→        scopeRequestRef = last.refNumber;
   387→      }
   388→      break;
   389→    }
   390→    case "specific_request": {
   391→      if (specificId) {
   392→        reqWhere.id = specificId;
   393→        const r = await db.trainingRequest.findUnique({
   394→          where: { id: specificId },
   395→          select: { refNumber: true },
   396→        });
   397→        scopeRequestRef = r?.refNumber ?? specificId;
   398→      }
   399→      break;
   400→    }
   401→    case "specific_course": {
   402→      if (specificId) {
   403→        scopeCourseId = specificId;
   404→        // A request is "for this course" if its primary courseId matches OR
   405→        // any of its requestCourses.courseId matches.
   406→        reqWhere.OR = [
   407→          { courseId: specificId },
   408→          { requestCourses: { some: { courseId: specificId, deletedAt: null } } },
   409→        ];
   410→        const c = await db.course.findUnique({ where: { id: specificId }, select: { title: true } });
   411→        scopeCourseTitle = c?.title ?? specificId;
   412→      }
   413→      break;
   414→    }
   415→    case "date_range": {
   416→      if (dateFrom || dateTo) {
   417→        const range: Record<string, Date> = {};
   418→        if (dateFrom) range.gte = new Date(dateFrom);
   419→        if (dateTo) {
   420→          const end = new Date(dateTo);
   421→          end.setHours(23, 59, 59, 999);
   422→          range.lte = end;
   423→        }
   424→        reqWhere.createdAt = range;
   425→      }
   426→      break;
   427→    }
   428→    case "all":
   429→    default:
   430→      // No additional filter — full company export
   431→      break;
   432→  }

   906→    // Phase 3: Trainee.idAttachmentUrl is no longer read. After Phase 2
   907→    // backfill, every URL that was in idAttachmentUrl also lives in

   910→    // Helper: extract a human-readable filename from the stored JSON.
   911→    // The upload handler stores both `filename` (random hex) and `originalName`
   912→    // (the real file name). Prefer originalName; fall back to filename; if
   913→    // neither is available, extract from the URL path.
   914→    function getDisplayName(doc: { filename?: string; originalName?: string; url?: string }): string {
   915→      if (doc.originalName && doc.originalName.length > 0) return doc.originalName;
   916→      if (doc.filename && doc.filename.length > 0 && !/^[a-f0-9]{32}\./.test(doc.filename)) return doc.filename;
   917→      // If filename is a random hex hash, try to extract from URL
   918→      const urlParts = doc.url?.split("/") ?? [];
   919→      const lastPart = urlParts[urlParts.length - 1] ?? "attachment";
   920→      // If even the URL part is a hex hash, show a generic name with extension
   921→      if (/^[a-f0-9]{32}\./.test(lastPart)) {
   922→        const ext = lastPart.split(".").pop() ?? "file";
   923→        return `Attachment.${ext}`;
   924→      }
   925→      return lastPart;
   926→    }
