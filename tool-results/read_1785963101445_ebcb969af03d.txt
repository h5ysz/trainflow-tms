   234→   234→// ── Phase 1: documents[] is the single source of truth ────────────────────
   235→   235→// The legacy `idAttachmentUrl` field is no longer written by any client
   236→   236→// callback. All ID uploads go into `documents[]` as `{type:"id"}` entries.
   237→   237→// These helpers locate the ID-type document within documents[] for display,
   238→   238→// stats, and filtering — they never read `idAttachmentUrl`.
   239→   239→const ID_DOC_TYPES: TraineeDocument["type"][] = ["id", "iqama"];
   240→   240→
   241→   241→function getIdDocument(row: TraineeEntry): TraineeDocument | undefined {
   242→   242→  return row.documents.find((d) => ID_DOC_TYPES.includes(d.type));
   243→   243→}
   244→   244→
   245→   245→function hasIdDocument(row: TraineeEntry): boolean {
   246→   246→  return getIdDocument(row) !== undefined;
   247→   247→}
   248→   248→
   249→   249→// Phase 1 — normalize legacy `idAttachmentUrl` into documents[] on load.
   250→   250→//
   251→   251→// When loading existing trainee data from the API (which still returns the
   252→   252→// legacy column for backward compat), fold any `idAttachmentUrl` value into
   253→   253→// `documents[]` as a `{type:"id"}` entry IF that URL is not already
   254→   254→// represented. This ensures the in-memory TraineeEntry state is always
   255→   255→// consistent with the new single-source-of-truth model, regardless of
   256→   256→// whether the DB row has been backfilled yet (Phase 2).
   257→   257→//
   258→   258→// After Phase 2 backfill completes, this becomes a no-op for all rows
   259→   259→// (the URL will already be in documents[]).
   260→   260→export function normalizeIdAttachmentIntoDocuments(
   261→   261→  idAttachmentUrl: string | null | undefined,
   262→   262→  documents: TraineeDocument[],
   263→   263→): TraineeDocument[] {
   264→   264→  if (!idAttachmentUrl) return documents;
   265→   265→  const alreadyPresent = documents.some((d) => d.url === idAttachmentUrl);
   266→   266→  if (alreadyPresent) return documents;
   267→   267→  const syntheticDoc: TraineeDocument = {
   268→   268→    url: idAttachmentUrl,
   269→   269→    filename: idAttachmentUrl.split("/").pop() ?? "id-attachment",
   270→   270→    type: "id",
   271→   271→    uploadedAt: new Date().toISOString(),
   272→   272→  };
   273→   273→  return [...documents, syntheticDoc];
   274→   274→}

   276→   276→function validateRow(row: TraineeEntry, allIds: Map<string, number>, selfIndex: number): TraineeEntry {
   277→   277→  const errors: string[] = [];
   278→   278→  if (!row.fullName.trim()) errors.push("Missing name");
   279→   279→  if (!row.nationalId.trim()) errors.push("Missing national ID");
   280→   280→  if (row.nationalId.trim()) {
   281→   281→    const firstSeen = allIds.get(row.nationalId.trim());
   282→   282→    if (firstSeen !== undefined && firstSeen !== selfIndex) {
   283→   283→      errors.push(`Duplicate national ID (first at row ${firstSeen + 1})`);
   284→   284→    }
   285→   285→  }
   286→   286→  return { ...row, valid: errors.length === 0, errors };
   287→   287→}

   289→   289→function recomputeValidity(rows: TraineeEntry[]): TraineeEntry[] {
   290→   290→  // Build a nationalId → first-index map, then re-run validation per row.
   291→   291→  const idMap = new Map<string, number>();
   292→   292→  rows.forEach((r, i) => {
   293→   293→    const id = r.nationalId.trim();
   294→   294→    if (id && !idMap.has(id)) idMap.set(id, i);
   295→   295→  });
   296→   296→  return rows.map((r, i) => validateRow(r, idMap, i));
   297→   297→}
