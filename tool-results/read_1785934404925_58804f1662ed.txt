    51→function RecentOperations({ logs, loading }: { logs: ImportExportLogEntry[]; loading: boolean }) {
    52→  const { t } = useI18n();
    53→  if (loading) {
    54→    return (
    55→      <div className="flex items-center justify-center py-3 gap-2 text-[10px] text-muted-foreground">
    56→        <Loader2 className="h-3 w-3 animate-spin" /> {t("misc.loading") || "Loading..."}
    57→      </div>
    58→    );
    59→  }
    60→  if (logs.length === 0) {
    61→    return (
    62→      <div className="text-center py-3 text-[10px] text-muted-foreground">
    63→        {t("requests.noRecentOps")}
    64→      </div>
    65→    );
    66→  }
    67→  return (
    68→    <div className="space-y-1 max-h-32 overflow-y-auto">
    69→      {logs.slice(0, 10).map((log) => (
    70→        <div key={log.id} className="flex items-center gap-2 text-[10px] py-1 px-2 rounded hover:bg-muted/40">
    71→          {log.type === "IMPORT" ? (
    72→            <Upload className="h-3 w-3 text-primary shrink-0" />
    73→          ) : (
    74→            <Download className="h-3 w-3 text-primary shrink-0" />
    75→          )}
    76→          <span className="font-medium shrink-0">{log.type}</span>
    77→          {log.requestRef && <span className="font-mono text-muted-foreground">{log.requestRef}</span>}
    78→          {log.courseName && <span className="text-muted-foreground truncate">· {log.courseName}</span>}
    79→          <span className="text-muted-foreground ms-auto shrink-0">
    80→            {new Date(log.createdAt).toLocaleString()}
    81→          </span>
    82→          {log.status === "FAILED" ? (
    83→            <Badge variant="destructive" className="text-[8px] h-3.5 px-1">{t("misc.error") || "Failed"}</Badge>
    84→          ) : (
    85→            <Badge variant="outline" className="text-[8px] h-3.5 px-1 text-emerald-600">{t("misc.success") || "OK"}</Badge>
    86→          )}
    87→        </div>
    88→      ))}
    89→    </div>
    90→  );
    91→}

    93→// ─── Hook: load + refresh logs ──────────────────────────────────────────────
    94→
    95→function useRecentLogs() {
    96→  const [logs, setLogs] = React.useState<ImportExportLogEntry[]>([]);
    97→  const [loading, setLoading] = React.useState(false);
    98→
    99→  const refresh = React.useCallback(async () => {
   100→    setLoading(true);
   101→    try {
   102→      const data = await api.get<ImportExportLogEntry[]>("/import-export-logs");
   103→      setLogs(data);
   104→    } catch { /* ignore */ } finally {
   105→      setLoading(false);
   106→    }
   107→  }, []);
   108→
   109→  const addLog = React.useCallback(async (entry: {
   110→    type: string; source: string; requestRef?: string; courseName?: string;
   111→    itemCount?: number; status?: string; errorMessage?: string;
   112→  }) => {
   113→    try {
   114→      await api.post("/import-export-logs", entry);
   115→      await refresh();
   116→    } catch { /* ignore */ }
   117→  }, [refresh]);
   118→
   119→  React.useEffect(() => { refresh(); }, [refresh]);
   120→
   121→  return { logs, loading, refresh, addLog };
   122→}

   398→export function ExportDialog({ open, onOpenChange }: ExportDialogProps) {

   400→  const { toast } = useToast();
   401→  const { logs, loading: logsLoading, addLog } = useRecentLogs();
   402→  const [scope, setScope] = React.useState<"last" | "specific_request" | "specific_course" | "date_range" | "all">("last");
   403→  const [items, setItems] = React.useState<Set<string>>(new Set(["requests", "trainees"]));

   405→  const [dateFrom, setDateFrom] = React.useState("");
   406→  const [dateTo, setDateTo] = React.useState("");
   407→  const [exporting, setExporting] = React.useState(false);
   408→  const [progress, setProgress] = React.useState(0);
   409→
   410→  // ── Course picker state (for specific_course scope) ──
   411→  const [courses, setCourses] = React.useState<SearchableSelectOption[]>([]);
   412→  const [coursesLoading, setCoursesLoading] = React.useState(false);
   413→  const [selectedCourseId, setSelectedCourseId] = React.useState("");
   414→
   415→  // ── Training Request picker state (for specific_request scope) ──
   416→  const [requests, setRequests] = React.useState<SearchableSelectOption[]>([]);
   417→  const [requestsLoading, setRequestsLoading] = React.useState(false);
   418→  const [selectedRequestId, setSelectedRequestId] = React.useState("");
   419→
   420→  // Fetch company courses when specific_course scope is selected
   421→  React.useEffect(() => {
   422→    if (scope === "specific_course" && courses.length === 0 && !coursesLoading) {
   423→      setCoursesLoading(true);
   424→      // /api/courses returns { success, data: Course[] } — api.get unwraps to `data`
   425→      // which is the array directly (not { items: [...] }).
   426→      api.get<Array<{ id: string; title: string; titleAr?: string; code?: string; refNumber?: string }>>("/courses?pageSize=200")
   427→        .then((arr) => {
   428→          const list = Array.isArray(arr) ? arr : [];
   429→          setCourses(list.map((c) => ({
   430→            value: c.id,
   431→            label: (locale === "ar" && c.titleAr ? c.titleAr : c.title) || c.refNumber || c.id,
   432→            description: [c.code, c.refNumber].filter(Boolean).join(" · ") || undefined,
   433→            keywords: `${c.title} ${c.titleAr ?? ""} ${c.code ?? ""} ${c.refNumber ?? ""}`,
   434→          })));
   435→        })
   436→        .catch(() => {
   437→          // Silent fail — picker will just be empty
   438→        })
   439→        .finally(() => setCoursesLoading(false));
   440→    }
   441→  }, [scope, courses.length, coursesLoading]);
   442→
   443→  // Fetch company training requests when specific_request scope is selected
   444→  React.useEffect(() => {
   445→    if (scope === "specific_request" && requests.length === 0 && !requestsLoading) {
   446→      setRequestsLoading(true);
   447→      // /api/requests returns { success, data: TrainingRequest[] } — api.get unwraps to `data`.
   448→      api.get<Array<{
   449→        id: string; refNumber: string; status: string; courseTitle: string | null;
   450→        traineeCount: number; createdAt: string;
   451→      }>>("/requests?pageSize=200")
   452→        .then((arr) => {
   453→          const list = Array.isArray(arr) ? arr : [];
   454→          setRequests(list.map((r) => ({
   455→            value: r.id,
   456→            label: r.refNumber,
   457→            description: [
   458→              r.courseTitle,
   459→              `${r.traineeCount} ${locale === "ar" ? "متدرب" : "trainees"}`,
   460→              r.status,
   461→            ].filter(Boolean).join(" · "),
   462→            keywords: `${r.refNumber} ${r.courseTitle ?? ""} ${r.status}`,
   463→          })));
   464→        })
   465→        .catch(() => {
   466→          // Silent fail — picker will just be empty
   467→        })
   468→        .finally(() => setRequestsLoading(false));
   469→    }
   470→  }, [scope, requests.length, requestsLoading, locale]);
   471→
   472→  // Reset selection when scope changes (so stale IDs don't leak across scopes)
   473→  React.useEffect(() => {
   474→    setSelectedCourseId("");
   475→    setSelectedRequestId("");
   476→    // Don't reset dateFrom/dateTo — they survive scope switching
   477→  }, [scope]);
   478→
   479→  const EXPORT_ITEMS = [
   480→    { key: "requests", label: t("requests.itemRequests") },
   481→    { key: "trainees", label: t("requests.itemTrainees") },
   482→    { key: "attendance", label: t("requests.itemAttendance") },
   483→    { key: "results", label: t("requests.itemResults") },
   484→    { key: "evaluations", label: t("requests.itemEvaluations") },
   485→    { key: "certificates", label: t("requests.itemCertificates") },
   486→    { key: "invoices", label: t("requests.itemInvoices") },
   487→    { key: "attachments", label: t("requests.itemAttachments") },
   488→  ];
   489→
   490→  const scopeOptions = [
   491→    { value: "last", label: t("requests.scopeLast") },
   492→    { value: "specific_request", label: t("requests.scopeSpecificRequest") },
   493→    { value: "specific_course", label: t("requests.scopeSpecificCourse") },
   494→    { value: "date_range", label: t("requests.scopeDateRange") },
   495→    { value: "all", label: t("requests.scopeAll") },
   496→  ] as const;
   497→
   498→  function toggleItem(key: string) {
   499→    setItems((prev) => {
   500→      const next = new Set(prev);
   501→      if (next.has(key)) next.delete(key);
   502→      else next.add(key);
   503→      return next;
   504→    });
   505→  }
   506→
   507→  // ── Validation: which scopes require an additional selection? ──
   508→  const needsSelection = scope === "specific_request" || scope === "specific_course";
   509→  const needsDateRange = scope === "date_range";
   510→  const selectionMissing = needsSelection
   511→    ? (scope === "specific_request" ? !selectedRequestId : !selectedCourseId)
   512→    : false;
   513→  const dateRangeMissing = needsDateRange && (!dateFrom || !dateTo);
   514→
   515→  const canExport = items.size > 0 && !selectionMissing && !dateRangeMissing && !exporting;
   516→
   517→  async function handleExport() {
   518→    if (items.size === 0) {
   519→      toast({ title: t("misc.error"), description: t("requests.selectAtLeastOne"), variant: "destructive" });
   520→      return;
   521→    }
   522→    if (scope === "specific_course" && !selectedCourseId) {
   523→      toast({ title: t("misc.error"), description: (t("requests.selectCoursePrompt") || "Please select a course"), variant: "destructive" });
   524→      return;
   525→    }
   526→    if (scope === "specific_request" && !selectedRequestId) {
   527→      toast({ title: t("misc.error"), description: (t("requests.selectRequestPrompt") || "Please select a request"), variant: "destructive" });
   528→      return;
   529→    }
   530→    if (scope === "date_range" && (!dateFrom || !dateTo)) {
   531→      toast({ title: t("misc.error"), description: (t("requests.selectDateRangePrompt") || "Please select both dates"), variant: "destructive" });
   532→      return;
   533→    }
   534→    setExporting(true);
   535→    setProgress(10);
   536→    try {
   537→      const params = new URLSearchParams({ scope, format, items: Array.from(items).join(","), locale });
   538→      if (dateFrom) params.set("dateFrom", dateFrom);
   539→      if (dateTo) params.set("dateTo", dateTo);
   540→      if (scope === "specific_course" && selectedCourseId) {
   541→        params.set("specificId", selectedCourseId);
   542→      }
   543→      if (scope === "specific_request" && selectedRequestId) {
   544→        params.set("specificId", selectedRequestId);
   545→      }
   546→      setProgress(50);
   547→      window.open(`/api/export/company-data?${params.toString()}`, "_blank");
   548→      setProgress(100);
   549→      toast({ title: t("misc.success"), description: t("requests.exportStarted") });
   550→      await addLog({ type: "EXPORT", source: "EXPORT", itemCount: items.size, status: "SUCCESS" });
   551→      onOpenChange(false);
   552→    } catch (e) {
   553→      const msg = e instanceof ApiError ? e.message : t("requests.exportFailed");
   554→      toast({ title: t("misc.error"), description: msg, variant: "destructive" });
   555→      await addLog({ type: "EXPORT", source: "EXPORT", status: "FAILED", errorMessage: msg });
   556→    } finally {
   557→      setExporting(false);
   558→      setProgress(0);
   559→    }
   560→  }
   561→
   562→  return (
   563→    <Dialog open={open} onOpenChange={onOpenChange}>
   564→      <DialogContent className="max-w-2xl max-h-[85vh] overflow-hidden p-0 gap-0 flex flex-col">
   565→        <DialogHeader className="p-5 border-b shrink-0">
   566→          <DialogTitle className="flex items-center gap-2">
   567→            <Download className="h-5 w-5 text-primary" />
   568→            {t("requests.export")}
   569→          </DialogTitle>
   570→          <DialogDescription className="text-xs">{t("requests.exportDesc")}</DialogDescription>
   571→        </DialogHeader>
   572→
   573→        <div className="p-5 overflow-y-auto flex-1 min-h-0 space-y-5">
   574→          {/* Scope — dynamic field rendering based on selected scope */}
   575→          <div>
   576→            <div className="text-xs font-semibold text-muted-foreground uppercase tracking-wide mb-2">{t("requests.scope")}</div>
   577→            <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   578→              {scopeOptions.map((opt) => (
   579→                <label key={opt.value}
   580→                  className={`flex items-center gap-2 rounded-md border p-2 cursor-pointer text-xs transition-colors ${
   581→                    scope === opt.value ? "border-primary bg-primary/5" : "hover:bg-muted/30"
   582→                  }`}>
   583→                  <input type="radio" name="export-scope" checked={scope === opt.value} onChange={() => setScope(opt.value)} className="accent-primary" />
   584→                  {opt.label}
   585→                </label>
   586→              ))}
   587→            </div>
   588→
   589→            {/* ── Dynamic field area: only renders for scopes that need extra input ── */}
   590→            {scope === "specific_request" && (
   591→              <div className="mt-3 space-y-1.5">
   592→                <label className="text-[10px] font-semibold text-muted-foreground uppercase tracking-wide">
   593→                  {t("requests.selectRequestPrompt") || "Select a request"}
   594→                </label>
   595→                <SearchableSelect
   596→                  value={selectedRequestId}
   597→                  onChange={setSelectedRequestId}
   598→                  options={requests}
   599→                  loading={requestsLoading}
   600→                  placeholder={t("requests.selectRequestPrompt") || "Select a request…"}
   601→                  searchPlaceholder={t("requests.searchRequestPlaceholder") || "Search by request # or course…"}
   602→                  emptyText={t("requests.noRequestsAvailable") || "No requests available"}
   603→                />
   604→                {requests.length === 0 && !requestsLoading && (
   605→                  <div className="text-[10px] text-muted-foreground">
   606→                    {t("requests.noRequestsAvailable") || "No requests available for this company"}
   607→                  </div>
   608→                )}
   609→              </div>
   610→            )}
   611→
   612→            {scope === "specific_course" && (
   613→              <div className="mt-3 space-y-1.5">
   614→                <label className="text-[10px] font-semibold text-muted-foreground uppercase tracking-wide">
   615→                  {t("requests.selectCoursePrompt") || "Select a course"}
   616→                </label>
   617→                <SearchableSelect
   618→                  value={selectedCourseId}
   619→                  onChange={setSelectedCourseId}
   620→                  options={courses}
   621→                  loading={coursesLoading}
   622→                  placeholder={t("requests.selectCoursePrompt") || "Select a course…"}
   623→                  searchPlaceholder={t("requests.searchCoursePlaceholder") || "Search by course name or code…"}
   624→                  emptyText={t("requests.noCoursesAvailable") || "No courses available"}
   625→                />
   626→                {courses.length === 0 && !coursesLoading && (
   627→                  <div className="text-[10px] text-muted-foreground">
   628→                    {t("requests.noCoursesAvailable") || "No courses available for this company"}
   629→                  </div>
   630→                )}
   631→              </div>
   632→            )}
   633→
   634→            {scope === "date_range" && (
   635→              <div className="mt-3 grid grid-cols-2 gap-2">
   636→                <div className="space-y-1">
   637→                  <label className="text-[10px] font-semibold text-muted-foreground uppercase tracking-wide block">
   638→                    {t("requests.dateFrom") || "From date"}
   639→                  </label>
   640→                  <Input type="date" value={dateFrom} onChange={(e) => setDateFrom(e.target.value)} className="text-xs h-9" />
   641→                </div>
   642→                <div className="space-y-1">
   643→                  <label className="text-[10px] font-semibold text-muted-foreground uppercase tracking-wide block">
   644→                    {t("requests.dateTo") || "To date"}
   645→                  </label>
   646→                  <Input type="date" value={dateTo} onChange={(e) => setDateTo(e.target.value)} className="text-xs h-9" />
   647→                </div>
   648→              </div>
   649→            )}
   650→
   651→            {/* last + all: no extra fields */}
   652→            {(scope === "last" || scope === "all") && (
   653→              <div className="mt-3 text-[10px] text-muted-foreground italic">
   654→                {scope === "last"
   655→                  ? (t("requests.scopeLastHint") || "The most recent request will be exported automatically.")
   656→                  : (t("requests.scopeAllHint") || "All company data will be exported.")}
   657→              </div>
   658→            )}
   659→          </div>
   660→
   661→          {/* Items */}
   662→          <div>
   663→            <div className="flex items-center justify-between mb-2">
   664→              <div className="text-xs font-semibold text-muted-foreground uppercase tracking-wide">{t("requests.itemsToExport")}</div>
   665→              <div className="flex gap-1">
   666→                <Button type="button" variant="ghost" size="sm" className="text-xs h-7" onClick={() => setItems(new Set(EXPORT_ITEMS.map(i => i.key)))}>{t("requests.selectAll")}</Button>
   667→                <Button type="button" variant="ghost" size="sm" className="text-xs h-7" onClick={() => setItems(new Set())}>{t("requests.clear")}</Button>
   668→              </div>
   669→            </div>
   670→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   671→              {EXPORT_ITEMS.map((item) => (
   672→                <label key={item.key} className="flex items-center gap-2 rounded-md border p-2 cursor-pointer hover:bg-muted/30">
   673→                  <Checkbox checked={items.has(item.key)} onCheckedChange={() => toggleItem(item.key)} />
   674→                  <span className="text-xs">{item.label}</span>
   675→                </label>
   676→              ))}
   677→            </div>
   678→          </div>
   679→
   680→          {/* Format */}
   681→          <div>
   682→            <div className="text-xs font-semibold text-muted-foreground uppercase tracking-wide mb-2">{t("requests.format")}</div>
   683→            <div className="flex gap-2">
   684→              {([
   685→                { value: "excel", label: "Excel", icon: FileSpreadsheet },
   686→                { value: "pdf", label: "PDF", icon: FileText },
   687→                { value: "zip", label: t("requests.zipWithAttachments"), icon: FileArchive },
   688→              ] as const).map((opt) => (
   689→                <label key={opt.value}
   690→                  className={`flex items-center gap-2 rounded-md border p-2.5 cursor-pointer text-xs transition-colors ${
   691→                    format === opt.value ? "border-primary bg-primary/5" : "hover:bg-muted/30"
   692→                  }`}>
   693→                  <input type="radio" name="export-format" checked={format === opt.value} onChange={() => setFormat(opt.value)} className="accent-primary" />
   694→                  <opt.icon className="h-3.5 w-3.5" />
   695→                  {opt.label}
   696→                </label>
   697→              ))}
   698→            </div>
   699→          </div>
   700→
   701→          {exporting && (
   702→            <div className="space-y-1">
   703→              <Progress value={progress} className="h-1.5" />
   704→              <div className="text-[10px] text-muted-foreground text-center">{t("requests.preparingExport")}</div>
   705→            </div>
   706→          )}
   707→        </div>
   708→
   709→        <DialogFooter className="p-4 border-t bg-muted/30 shrink-0">
   710→          <Button variant="outline" onClick={() => onOpenChange(false)}>{t("action.cancel")}</Button>
   711→          <Button onClick={handleExport} disabled={!canExport} title={!canExport && selectionMissing ? (t("requests.selectFirstPrompt") || "Select an item first") : undefined}>
   712→            {exporting ? <Loader2 className="h-4 w-4 me-1.5 animate-spin" /> : <FileDown className="h-4 w-4 me-1.5" />}
   713→            {t("requests.exportBtn")}
   714→          </Button>
   715→        </DialogFooter>
   716→      </DialogContent>
   717→    </Dialog>
   718→  );
   719→}
