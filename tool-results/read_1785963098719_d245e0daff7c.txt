   210→function makeId(): string {
   211→  // crypto.randomUUID is available in all evergreen browsers + Node 19+.
   212→  if (typeof crypto !== "undefined" && "randomUUID" in crypto) {
   213→    return crypto.randomUUID();
   214→  }
   215→  return `t-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`;
   216→}

   234→// ── Phase 1: documents[] is the single source of truth ────────────────────
   235→// The legacy `idAttachmentUrl` field is no longer written by any client
   236→// callback. All ID uploads go into `documents[]` as `{type:"id"}` entries.
   237→// These helpers locate the ID-type document within documents[] for display,
   238→// stats, and filtering — they never read `idAttachmentUrl`.
   239→const ID_DOC_TYPES: TraineeDocument["type"][] = ["id", "iqama"];
   240→
   241→function getIdDocument(row: TraineeEntry): TraineeDocument | undefined {
   242→  return row.documents.find((d) => ID_DOC_TYPES.includes(d.type));
   243→}
   244→
   245→function hasIdDocument(row: TraineeEntry): boolean {
   246→  return getIdDocument(row) !== undefined;
   247→}
   248→
   249→// Phase 1 — normalize legacy `idAttachmentUrl` into documents[] on load.
   250→//
   251→// When loading existing trainee data from the API (which still returns the
   252→// legacy column for backward compat), fold any `idAttachmentUrl` value into
   253→// `documents[]` as a `{type:"id"}` entry IF that URL is not already
   254→// represented. This ensures the in-memory TraineeEntry state is always
   255→// consistent with the new single-source-of-truth model, regardless of
   256→// whether the DB row has been backfilled yet (Phase 2).
   257→//
   258→// After Phase 2 backfill completes, this becomes a no-op for all rows
   259→// (the URL will already be in documents[]).
   260→export function normalizeIdAttachmentIntoDocuments(
   261→  idAttachmentUrl: string | null | undefined,
   262→  documents: TraineeDocument[],
   263→): TraineeDocument[] {
   264→  if (!idAttachmentUrl) return documents;
   265→  const alreadyPresent = documents.some((d) => d.url === idAttachmentUrl);
   266→  if (alreadyPresent) return documents;
   267→  const syntheticDoc: TraineeDocument = {
   268→    url: idAttachmentUrl,
   269→    filename: idAttachmentUrl.split("/").pop() ?? "id-attachment",
   270→    type: "id",
   271→    uploadedAt: new Date().toISOString(),
   272→  };
   273→  return [...documents, syntheticDoc];
   274→}
   275→
   276→function validateRow(row: TraineeEntry, allIds: Map<string, number>, selfIndex: number): TraineeEntry {
   277→  const errors: string[] = [];
   278→  if (!row.fullName.trim()) errors.push("Missing name");
   279→  if (!row.nationalId.trim()) errors.push("Missing national ID");
   280→  if (row.nationalId.trim()) {
   281→    const firstSeen = allIds.get(row.nationalId.trim());
   282→    if (firstSeen !== undefined && firstSeen !== selfIndex) {
   283→      errors.push(`Duplicate national ID (first at row ${firstSeen + 1})`);
   284→    }
   285→  }
   286→  return { ...row, valid: errors.length === 0, errors };
   287→}
   288→
   289→function recomputeValidity(rows: TraineeEntry[]): TraineeEntry[] {
   290→  // Build a nationalId → first-index map, then re-run validation per row.
   291→  const idMap = new Map<string, number>();
   292→  rows.forEach((r, i) => {
   293→    const id = r.nationalId.trim();
   294→    if (id && !idMap.has(id)) idMap.set(id, i);
   295→  });
   296→  return rows.map((r, i) => validateRow(r, idMap, i));
   297→}

   318→  const [previewFullscreen, setPreviewFullscreen] = React.useState(false);
   319→  const [bulkUpload, setBulkUpload] = React.useState<BulkUploadState>({
   320→    phase: "idle", total: 0, done: 0, matched: 0, unmatched: [],
   321→  });
   322→  const [pasteText, setPasteText] = React.useState("");
   323→
   324→  // Refs for the virtualized scroll container.
   325→  const scrollRef = React.useRef<HTMLDivElement | null>(null);
   326→  const [scrollTop, setScrollTop] = React.useState(0);
   327→
   328→  // ── BUG-009/011 fix: force re-render after mount ─────────────────────
   329→  // The virtualization calculates visibleSlice from scrollTop (0) and
   330→  // currentTableHeight (constant). On first render inside a Dialog, the
   331→  // TabsContent mounts lazily and the scroll container's layout isn't
   332→  // settled until after paint. This causes the table body to appear empty
   333→  // until the user interacts (resize, scroll, tab switch).
   334→  //
   335→  // Fix: use useLayoutEffect to synchronously set a "mounted" flag BEFORE
   336→  // the browser paints. This triggers a second render with the correct
   337→  // layout, so the rows appear immediately.
   338→  const [mounted, setMounted] = React.useState(false);
   339→  React.useLayoutEffect(() => {
   340→    setMounted(true);
   341→  }, []);
   342→
   343→  // Also re-trigger on trainees change (e.g. after Excel import the table
   344→  // should immediately show the new rows without requiring a scroll).
   345→  React.useLayoutEffect(() => {
   346→    if (mounted && scrollRef.current) {
   347→      // Reset scroll to top when trainees change so the virtualization
   348→      // re-calculates from scrollTop=0.
   349→      if (scrollRef.current.scrollTop !== 0) {
   350→        scrollRef.current.scrollTop = 0;
   351→      }
   352→      setScrollTop(0);
   353→    }
   354→  }, [trainees, mounted]);
   355→
   356→  // ─── Derived stats (memoized) ──────────────────────────────────────────
   357→  const stats = React.useMemo(() => {
   358→    const idCounts = new Map<string, number>();
   359→    for (const r of trainees) {
   360→      const id = r.nationalId.trim();
   361→      if (id) idCounts.set(id, (idCounts.get(id) ?? 0) + 1);
   362→    }
   363→    const total = trainees.length;
   364→    const valid = trainees.filter((r) => r.valid).length;
   365→    const invalid = total - valid;
   366→    const duplicates = trainees.filter((r) => {
   367→      const id = r.nationalId.trim();
   368→      return id && (idCounts.get(id) ?? 0) > 1;
   369→    }).length;
   370→    const missingAttachment = trainees.filter(
   371→      (r) => r.fullName.trim() && r.nationalId.trim() && !hasIdDocument(r)
   372→    ).length;
   373→    return { total, valid, invalid, duplicates, missingAttachment };
   374→  }, [trainees]);
   375→

   377→  const filtered = React.useMemo(() => {
   378→    const q = search.trim().toLowerCase();

   381→      const id = r.nationalId.trim();
   382→      if (id) idCounts.set(id, (idCounts.get(id) ?? 0) + 1);
   383→    }

   385→      if (q) {
   386→        const hay = `${r.fullName} ${r.nationalId} ${r.jobTitle} ${r.nationality}`.toLowerCase();
   387→        if (!hay.includes(q)) return false;

   390→        case "valid": return r.valid;
   391→        case "invalid": return !r.valid;
   392→        case "duplicates": {
   393→          const id = r.nationalId.trim();
   394→          return id ? (idCounts.get(id) ?? 0) > 1 : false;
   395→        }
   396→        case "missing-attachment":
   397→          return r.fullName.trim() !== "" && r.nationalId.trim() !== "" && !hasIdDocument(r);
   398→        default: return true;
   399→      }
   400→    });

   403→  // ─── Virtualization window ─────────────────────────────────────────────
   404→  // In Full Screen mode the table should fill the viewport. We reserve:
   405→  //   - ~64 px for the section padding (top + bottom of fixed inset-0)
   406→  //   - ~48 px for the tabs row (tabs + Full Screen button)
   407→  // ── BUG-009 fix: measure the actual scroll container height ───────────
   408→  // The constant VISIBLE_TABLE_HEIGHT (480) is a fallback. The real height

   413→  const [measuredHeight, setMeasuredHeight] = React.useState(0);

   415→    if (!scrollRef.current) return;

   417→      if (scrollRef.current) {
   418→        const h = scrollRef.current.clientHeight;
   419→        if (h > 0) setMeasuredHeight(h);

  1362→            <div className="text-xs text-muted-foreground">

  1381→            <div className="flex items-center gap-2 ms-auto">
  1382→              <Button variant="outline" onClick={() => { setPreviewOpen(false); setPreview(null); setExcelPendingFile(null); setPreviewFullscreen(false); }}>

  1384→              </Button>

  1391→                  // Allow import even with missing required columns — the user

  1396→              >
  1397→                <ArrowRight className="h-4 w-4 me-1.5" />{t("requests.import.confirm")}
  1398→              </Button>
  1399→            </div>
  1400→          </DialogFooter>
  1401→        </DialogContent>
  1402→      </Dialog>
  1403→    </div>

  1409→function StatCard({
  1410→  icon, label, value, active, tone, onClick, onJump,
  1411→}: {
  1412→  icon: React.ReactNode;
  1413→  label: string;

  1425→    : "";
  1426→
  1427→  return (
